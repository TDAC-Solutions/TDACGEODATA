# Borelog report rendering (app/borelog_report/pdf_renderer.py).
# The v6 script drew its log with matplotlib; the port keeps the same drawing

# numpy is matplotlib's own dependency -- the report module does not import it

# a result sheet is requested. reportlab is pure Python and installs on
# Termux; pinned to the version the engine PDFs were verified against.
reportlab==4.4.10

# The v6 rock-logging appendix (app/borelog_report/rock_appendix.py): the
# reportlab table is joined behind the matplotlib log pages with pypdf. Same
# lazy-import rule -- a preview request imports neither.
pypdf==6.17.0

# The v6 soil-profile narrative export
# (app/borelog_report/profile_docx.py). python-docx pulls lxml; pinned so a
# rebuild is reproducible.

lxml==6.1.3
